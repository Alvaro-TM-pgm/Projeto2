    <h2>Cadastro de Alunos</h2>
    
    <div class="form-group">
        <input type="text" id="nomeAluno" placeholder="Digite o nome do aluno">
        <button onclick="adicionarAluno()">Adicionar</button>
    </div>

    <table>
        <thead>
            <tr>
                <th>Nome do Aluno</th>
            </tr>
        </thead>
        <tbody id="corpoTabela">
            <!-- Lista será inserida aqui -->
        </tbody>
    </table>

    <script>
        const API_URL = "SUA_URL_DO_APPS_SCRIPT";

        // Carrega a lista ao abrir a página
        window.onload = buscarAlunos;

        async function buscarAlunos() {
            const tbody = document.getElementById('corpoTabela');
            tbody.innerHTML = "<tr><td>Carregando...</td></tr>";
            
            try {
                const response = await fetch(API_URL);
                const alunos = await response.json();
                
                tbody.innerHTML = "";
                alunos.forEach(nome => {
                    const tr = document.createElement('tr');
                    tr.innerHTML = `<td>${nome}</td>`;
                    tbody.appendChild(tr);
                });
            } catch (error) {
                console.error("Erro ao buscar:", error);
                tbody.innerHTML = "<tr><td>Erro ao carregar lista.</td></tr>";
            }
        }

        async function adicionarAluno() {
            const input = document.getElementById('nomeAluno');
            const nome = input.value.trim();
            
            if (!nome) return alert("Digite um nome!");

            try {
                // Usando modo 'no-cors' para envio simples se necessário, 
                // mas Apps Script com JSON geralmente aceita fetch normal
                await fetch(API_URL, {
                    method: 'POST',
                    mode: 'no-cors', // Evita problemas de pre-flight CORS
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ nome: nome })
                });

                input.value = "";
                // Como 'no-cors' não retorna o body, aguardamos um pouco e atualizamos
                setTimeout(buscarAlunos, 1000); 
            } catch (error) {
                console.error("Erro ao salvar:", error);
            }
        }
    </script>
